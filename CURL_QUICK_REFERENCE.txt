2️⃣  GENERATE SIMPLE IMAGE (Quick test)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  -H "Content-Type: application/json" \
  -d '{"text": "test image"}' \
  --output test.png

3️⃣  GENERATE WITH CUSTOM SETTINGS (Full control)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

curl -X POST "http://YOUR_SERVER_IP:15800/image/generate" \
  -H "Content-Type: application/json" \
  -d '{
    "text": "3D cute character with blue background",
    "model": "flux",
    "width": 1024,
    "height": 1024
  }' \
  --output my_image.png

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

4️⃣  GENERATE MULTIPLE IMAGES (Get metadata)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

    "width": 512,

5️⃣  RETRIEVE GENERATED IMAGE (By filename)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

🔧 TROUBLESHOOTING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Connection Refused:
  → Check if server is running: python run_api.py
  → Verify port 15800 is open in firewall

Timeout:
  → Add --max-time 300 to curl command
  → Try faster model (flux instead of sdxl)

Cannot Connect from Other Device:
  → Check server IP: hostname -I
  → Ensure API binds to 0.0.0.0 (not 127.0.0.1)
  → Open firewall: sudo ufw allow 15800/tcp
